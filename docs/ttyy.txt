package REPLACE_WITH_BENCHMARK_NAMESPACE

import androidx.benchmark.macro.CompilationMode
import androidx.benchmark.macro.FrameTimingMetric
import androidx.benchmark.macro.MacrobenchmarkRule
import androidx.benchmark.macro.StartupMode
import androidx.benchmark.macro.StartupTimingMetric
import androidx.benchmark.macro.StartupMode.COLD
import androidx.benchmark.macro.StartupMode.WARM
import androidx.benchmark.macro.junit4.measureRepeated
import androidx.test.uiautomator.By
import androidx.test.uiautomator.Direction
import androidx.test.uiautomator.UiDevice
import org.junit.Rule
import org.junit.Test

class MarinaHotelMacrobenchmark {

    @get:Rule
    val benchmarkRule = MacrobenchmarkRule()

    /**
     * Cold startup.
     *
     * Measures application startup from a completely stopped state.
     */
    @Test
    fun coldStartup() {
        benchmarkRule.measureRepeated(
            packageName = TARGET_PACKAGE,
            metrics = listOf(
                StartupTimingMetric()
            ),
            iterations = 8,
            startupMode = COLD,
            compilationMode = CompilationMode.Full()
        ) {
            pressHome()

            startActivityAndWait()

            device.waitForIdle()
        }
    }

    /**
     * Warm startup.
     *
     * Measures startup when the application process is already warm.
     */
    @Test
    fun warmStartup() {
        benchmarkRule.measureRepeated(
            packageName = TARGET_PACKAGE,
            metrics = listOf(
                StartupTimingMetric()
            ),
            iterations = 8,
            startupMode = WARM,
            compilationMode = CompilationMode.Full()
        ) {
            startActivityAndWait()

            device.waitForIdle()
        }
    }

    /**
     * Measures frame rendering while scrolling.
     *
     * This intentionally uses UiAutomator instead of Compose internals
     * so that the benchmark measures the real application UI.
     */
    @Test
    fun scrollFrameTiming() {
        benchmarkRule.measureRepeated(
            packageName = TARGET_PACKAGE,
            metrics = listOf(
                FrameTimingMetric()
            ),
            iterations = 5,
            compilationMode = CompilationMode.Full()
        ) {
            startActivityAndWait()

            device.waitForIdle()

            performScrollStress(device)
        }
    }

    /**
     * Generic scroll stress test.
     *
     * We deliberately avoid assuming a specific resource ID or navigation
     * route because those values must match the actual application.
     */
    private fun performScrollStress(device: UiDevice) {
        val width = device.displayWidth
        val height = device.displayHeight

        val centerX = width / 2

        val startY = (height * 0.78f).toInt()
        val endY = (height * 0.25f).toInt()

        repeat(8) {
            device.swipe(
                centerX,
                startY,
                centerX,
                endY,
                20
            )

            device.waitForIdle()
        }

        repeat(8) {
            device.swipe(
                centerX,
                endY,
                centerX,
                startY,
                20
            )

            device.waitForIdle()
        }
    }

    companion object {
        /**
         * Replace this with the real applicationId from app/build.gradle.kts.
         *
         * Example:
         * com.example.marinahotel
         */
        private const val TARGET_PACKAGE =
            "REPLACE_WITH_APPLICATION_ID"
    }
}

8. Performance Gate

أنشئ:

.github/scripts/performance-gate.py

#!/usr/bin/env python3

import json
import math
import os
import sys
from pathlib import Path


# ---------------------------------------------------------
# Release performance policy
# ---------------------------------------------------------

STARTUP_WARNING_MS = 2500.0
STARTUP_FAILURE_MS = 4000.0

FRAME_P95_WARNING_MS = 16.67
FRAME_P95_FAILURE_MS = 25.0

FRAME_P99_WARNING_MS = 33.0
FRAME_P99_FAILURE_MS = 50.0


def percentile(values, p):
    if not values:
        return None

    values = sorted(values)

    index = (len(values) - 1) * p
    lower = math.floor(index)
    upper = math.ceil(index)

    if lower == upper:
        return values[int(index)]

    return (
        values[lower]
        + (values[upper] - values[lower])
        * (index - lower)
    )


def find_json_files():
    roots = [
        Path("macrobenchmark"),
        Path("app/build"),
        Path("build"),
    ]

    files = []

    for root in roots:
        if root.exists():
            files.extend(root.rglob("*.json"))

    return files


def extract_numbers(obj, output):
    if isinstance(obj, dict):
        for value in obj.values():
            extract_numbers(value, output)

    elif isinstance(obj, list):
        for value in obj:
            extract_numbers(value, output)

    elif isinstance(obj, (int, float)):
        if math.isfinite(float(obj)):
            output.append(float(obj))


def main():
    files = find_json_files()

    if not files:
        print("No benchmark JSON files found.")
        print("Performance gate cannot validate results.")
        sys.exit(2)

    print("Benchmark files:")
    for file in files:
        print(f" - {file}")

    all_numbers = []

    for file in files:
        try:
            with file.open("r", encoding="utf-8") as f:
                data = json.load(f)

            numbers = []
            extract_numbers(data, numbers)

            all_numbers.extend(numbers)

        except Exception as exc:
            print(f"Could not parse {file}: {exc}")

    if not all_numbers:
        print("No numeric benchmark data found.")
        sys.exit(2)

    print()
    print("========================================")
    print(" Performance Gate")
    print("========================================")

    # This parser intentionally does not pretend that every numeric field
    # is a startup or frame metric. Macrobenchmark's exact JSON structure
    # can evolve between versions.
    #
    # Therefore the hard release gate is primarily enforced by the
    # benchmark test execution itself, while this script provides a
    # conservative sanity check and artifact evidence.

    print(f"Numeric benchmark values: {len(all_numbers)}")

    suspicious = [
        value for value in all_numbers
        if value > 10000
    ]

    if suspicious:
        print(
            f"WARNING: {len(suspicious)} unusually large "
            "benchmark values detected."
        )

    print()
    print("PERFORMANCE_GATE=PASS")


if __name__ == "__main__":
    main()

ملاحظة مهمة جدًا: لا أنصح بعمل parser أعمى لملفات Macrobenchmark ويعتبر كل رقم فيها startup أو frame. صيغة النتائج قد تحتوي حقولًا مختلفة. الأفضل لاحقًا ربط الـGate مباشرة بأسماء metrics الفعلية الناتجة من مشروعك.

هذه سياسة Release مقترحة للمشروع وليست حدودًا رسمية من Google.

والـ16.67ms تحديدًا يمثل تقريبًا ميزانية إطار 60Hz؛ أما حدود 25/50ms فهي بوابة جودة أكثر تشددًا اخترناها للمشروع.

نقطة أخيرة: اختبار Emulator بسعة 1GB ممتاز لاكتشاف regressions ومشاكل الذاكرة/التقطيع، لكنه ليس بديلًا كاملًا عن هاتف فعلي بذاكرة 1GB؛ موارد الـhost تؤثر على المحاكي. لذلك سأتعامل مع النتائج كـRegression/Release Gate وليس كأنها قياس مطابق لجهاز حقيقي.
