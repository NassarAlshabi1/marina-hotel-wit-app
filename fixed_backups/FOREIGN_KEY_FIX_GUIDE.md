  IntColumn get bookingLocalId => integer().nullable().references(Bookings, #id)();

// في جدول BookingNights

  // ... حقول أخرى
  
  // ✅ نفس التعديل:

#### الخطوة 2: Migration

أنشئ migration جديد في `lib/services/local_db.dart`:

```dart
@override
int get schemaVersion => 17; // زيادة رقم الإصدار

@override
MigrationStrategy get migration {
  return MigrationStrategy(
    onUpgrade: (Migrator m, int from, int to) async {
      if (from < 17) {
        // تعطيل Foreign Keys مؤقتاً
        await customStatement('PRAGMA foreign_keys = OFF');
        
        try {
          // إعادة إنشاء الجداول بـ schema الجديد
          await m.alterTable(TableMigration(payments));
          await m.alterTable(TableMigration(bookingNights));
        } finally {
          await customStatement('PRAGMA foreign_keys = ON');
        }
      }
    },
  );
}
```
